cmake_minimum_required(VERSION 3.22.1)
project(prometheus LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_SRC ${CMAKE_CURRENT_LIST_DIR}/../vendor/llama.cpp)
if(NOT EXISTS "${LLAMA_SRC}/include/llama.h")
    message(FATAL_ERROR "Missing vendored llama.cpp source: ${LLAMA_SRC}")
endif()
add_subdirectory(${LLAMA_SRC} ${CMAKE_CURRENT_BINARY_DIR}/llama)
add_library(prometheus SHARED native.cpp)
target_include_directories(prometheus PRIVATE ${LLAMA_SRC}/include ${LLAMA_SRC}/ggml/include)
target_link_libraries(prometheus llama android log)

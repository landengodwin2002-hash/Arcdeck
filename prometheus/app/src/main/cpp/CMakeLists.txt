cmake_minimum_required(VERSION 3.22.1)
# Inference must be optimized even when the Java APK is a debug test build.
# This follows the upstream llama.android library configuration.
set(CMAKE_BUILD_TYPE Release CACHE STRING "Optimized on-device inference" FORCE)
project(prometheus LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set_property(GLOBAL PROPERTY JOB_POOLS compile_pool=3 link_pool=1)
set(CMAKE_JOB_POOL_COMPILE compile_pool)
set(CMAKE_JOB_POOL_LINK link_pool)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_APP OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(LLAMA_SRC ${CMAKE_CURRENT_LIST_DIR}/../vendor/llama.cpp)
if(NOT EXISTS "${LLAMA_SRC}/tools/server/server.cpp")
    message(FATAL_ERROR "Run python3 scripts/fetch_runtime.py before building")
endif()
add_subdirectory(${LLAMA_SRC} ${CMAKE_CURRENT_BINARY_DIR}/llama)
add_library(prometheus SHARED native.cpp)
target_link_libraries(prometheus PRIVATE llama-server-impl android log)
target_link_options(prometheus PRIVATE "-Wl,-z,max-page-size=16384")
